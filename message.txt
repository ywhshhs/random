local function accept(value)
	if type(value)=="buffer"then local ok,text=pcall(bufferText,value);value=ok and text or nil end
	if type(value)~="string"then problem("unsupported response body",1);return end
	if #value~=expected then
		if #value<1024 and find(value,"Failed to send HTTP request.",1,true)then
			problem(find(value,"timed out",1,true)and"HTTP request timed out"or"HTTP request failed",2)
		else problem("received "..#value.." bytes; expected "..expected,2)end
		return
	end
	-- Eight bytes per Adler-32 step.
	local s1,s2,i=1.0,0.0,1
	while i+7<=#value do
		local a,b,c,d,e,f,g,h=byte(value,i,i+7)
		s2=(s2+8*s1+8*a+7*b+6*c+5*d+4*e+3*f+2*g+h)%65521
		s1=(s1+a+b+c+d+e+f+g+h)%65521;i=i+8
	end
	while i<=#value do s1=(s1+byte(value,i))%65521;s2=(s2+s1)%65521;i=i+1 end
	if s2*65536.0+s1~=668314204 then problem("runtime checksum mismatch",3);return end
	local ok,compiled,detail=pcall(compile,value,"=FlowAuthRuntime")
	if ok and type(compiled)=="function"then runtime=compiled;return true end
	reason=(ok and detail or compiled)or"compiler returned no function"
	return false
end
